import 'package:flutter/material.dart';
import 'package:window_manager/window_manager.dart';
import 'app_state.dart';
import 'desktop.dart';
import 'fmt.dart';
import 'jalali.dart';
import 'strings.dart';
import 'ui.dart';

/// ویجت دسکتاپ ویندوز: دو مربع کنار هم؛ تاریخ شمسی (چپ) و ساعت (راست). دو مدل: سفید و شفاف.
class DesktopWidget extends StatelessWidget {
  const DesktopWidget({super.key});

  @override
  Widget build(BuildContext context) {
    return ValueListenableBuilder<int>(
      valueListenable: Desktop.instance.widgetStyle,
      builder: (context, style, _) {
        final white = style == 0;
        final accent = context.accent;
        final shadow = white ? const <Shadow>[] : const [Shadow(color: Colors.black87, blurRadius: 8, offset: Offset(0, 2))];
        Widget square(Widget child) => Expanded(
              child: Center(
                child: AspectRatio(
                  aspectRatio: 1,
                  child: Container(
                    decoration: white ? BoxDecoration(color: Colors.white, borderRadius: BorderRadius.circular(26)) : null,
                    padding: const EdgeInsets.all(10),
                    child: FittedBox(child: child),
                  ),
                ),
              ),
            );
        return Scaffold(
          backgroundColor: Colors.transparent,
          body: DragToMoveArea(
            child: Stack(children: [
              Padding(
                padding: const EdgeInsets.all(6),
                child: LiveTime(builder: (context, now) {
                  final j = Jalali.fromDateTime(now);
                  final holiday = isHoliday(j);
                  final dayColor = white ? (holiday ? Colors.red.shade400 : accent) : Colors.white;
                  return Row(children: [
                    square(Column(mainAxisSize: MainAxisSize.min, children: [
                      Text('${j.weekdayName}  ${j.monthName}',
                          style: TextStyle(color: white ? Colors.black54 : Colors.white, fontSize: 14, fontWeight: FontWeight.bold, shadows: shadow)),
                      Text(nd(j.day), style: TextStyle(fontSize: 72, height: 1.0, fontWeight: FontWeight.bold, color: dayColor, shadows: shadow)),
                    ])),
                    const SizedBox(width: 8),
                    square(Ltr(
                      child: Text(fmtHM(now),
                          style: TextStyle(fontSize: 48, fontWeight: FontWeight.bold, color: white ? const Color(0xFF111111) : Colors.white, shadows: shadow)),
                    )),
                  ]);
                }),
              ),
              Positioned(
                top: 0,
                right: 0,
                child: Row(children: [
                  IconButton(
                    visualDensity: VisualDensity.compact,
                    tooltip: 'سفید / شفاف',
                    onPressed: () => Desktop.instance.toggleWidgetStyle(),
                    icon: const Icon(Icons.contrast, size: 16, color: Colors.grey),
                  ),
                  IconButton(
                    visualDensity: VisualDensity.compact,
                    tooltip: tr('widget_back'),
                    onPressed: () => Desktop.instance.exitWidget(),
                    icon: const Icon(Icons.open_in_full, size: 16, color: Colors.grey),
                  ),
                ]),
              ),
            ]),
          ),
        );
      },
    );
  }
}
